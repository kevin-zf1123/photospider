cmake_minimum_required(VERSION 3.21)
project(PhotospiderMultiOutputWorkflow LANGUAGES CXX)
if(NOT TARGET Photospider::kernel)
  find_package(Photospider 0.9 CONFIG REQUIRED COMPONENTS kernel)
endif()
add_executable(photospider_multi_output_workflow main.cpp)
target_link_libraries(photospider_multi_output_workflow PRIVATE Photospider::kernel)
target_compile_features(photospider_multi_output_workflow PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(photospider_multi_output_workflow PRIVATE -fno-fast-math -ffp-contract=off)
endif()
